815 template <class ID>
816 Vertex add_conditional_gate(
817 OpType type, const std::vector<Expr> &params, const std::vector<ID> &args,
818 const std::vector<ID> &bits, unsigned value,
819 std::optional<std::string> opgroup = std::nullopt) {
820 if (is_metaop_type(type)) {
821 throw CircuitInvalidity("Cannot add a conditional metaop.");
822 }
823 if (is_barrier_type(type)) {
824 throw CircuitInvalidity(
825 "Please use 'add_conditional_barrier' to add a conditional barrier "
826 "gate.");
827 }
828 Op_ptr cond = std::make_shared<Conditional>(
829 get_op_ptr(type, params, (unsigned)args.size()), (unsigned)bits.size(),
830 value);
831 std::vector<ID> new_args = bits;
832 new_args.insert(new_args.end(), args.begin(), args.end());
833 return add_op(cond, new_args, opgroup);
834 }

1400 if (opgroupsigs.find(opname) == opgroupsigs.end()) {
1401 return false;
1402 }
1403
1404 // Check signatures match
1405 op_signature_t sig = opgroupsigs[opname];

1407 throw CircuitInvalidity("Signature mismatch");

1411 BGL_FORALL_VERTICES(v, dag, DAG) {
1412 std::optional<std::string> v_opgroup = get_opgroup_from_Vertex(v);
1413 if (v_opgroup && v_opgroup.value() == opname) {
1414 to_replace.push_back(v);

1417
1418 unsigned sig_n_q = std::count(sig.begin(), sig.end(), EdgeType::Quantum);
1419 unsigned sig_n_c = std::count(sig.begin(), sig.end(), EdgeType::Classical);
1420 Circuit c(sig_n_q, sig_n_c);
1421 unit_vector_t args(sig_n_q + sig_n_c);
1422 for (unsigned i = 0; i < sig_n_q; i++) args[i] = Qubit(i);
1423 for (unsigned i = 0; i < sig_n_c; i++) args[sig_n_q + i] = Bit(i);
1424 c.add_box(to_insert, args, opname);
1425 for (const Vertex &v : to_replace) {
1426 substitute(c, v, VertexDeletion::Yes, OpGroupTransfer::Merge);

1657template <typename UnitA, typename UnitB>
1658bool Circuit::rename_units(const std::map<UnitA, UnitB> &qm) {
1659 // Can only work for Unit classes
1660 static_assert(std::is_base_of<UnitID, UnitA>::value);
1661 static_assert(std::is_base_of<UnitID, UnitB>::value);
1662 // Unit types must be related, so cannot rename e.g. Bits to Qubits
1663 static_assert(
1664 std::is_base_of<UnitA, UnitB>::value ||
1665 std::is_base_of<UnitB, UnitA>::value);
1666 std::map<UnitID, BoundaryElement> new_elems;
1667 bool modified = false;
1668 std::map<Bit, Bit> bm;
1669 for (const std::pair<const UnitA, UnitB> &pair : qm) {
1670 boundary_t::iterator found = boundary.get<TagID>().find(pair.first);
1671 if (found == boundary.get<TagID>().end()) {
1672 std::stringstream ss;
1673 ss << "unit " << pair.first.repr() << " not found in circuit";
1674 tket_log()->warn(ss.str());
1675 continue;
1676 }
1677
1678 opt_reg_info_t target_reg_info = get_reg_info(pair.second.reg_name());
1679 if (target_reg_info) {
1680 if (target_reg_info.value().first != found->type())
1681 throw CircuitInvalidity(
1682 "Incompatible registers: " + pair.first.reg_name() + " and " +
1683 pair.second.reg_name());
1684 if (target_reg_info.value().second != pair.second.reg_dim())
1685 throw CircuitInvalidity(
1686 "Existing register " + pair.second.reg_name() +
1687 " cannot support id: " + pair.second.repr());
1688 }
1689 std::pair<std::map<UnitID, BoundaryElement>::iterator, bool> inserted =
1690 new_elems.insert({pair.second, {pair.second, found->in_, found->out_}});
1691 if (!inserted.second)
1692 throw CircuitInvalidity(
1693 "Mapping two units to the same id: " + pair.second.repr());
1694 modified = true;
1695 boundary.erase(found);
1696 if (pair.first.type() == UnitType::Bit) {
1697 bm.insert({Bit(pair.first), Bit(pair.second)});
1698 }
1699 }
1700 for (const std::pair<const UnitID, BoundaryElement> &pair : new_elems) {
1701 std::pair<boundary_t::iterator, bool> added = boundary.insert(pair.second);
1702 if (!added.second)
1703 throw CircuitInvalidity(
1704 "Unit already exists in circuit: " + pair.first.repr());
1705 TKET_ASSERT(modified);
1706 }
1707 return modified;
1708}

1715template <class ID>
1716Vertex Circuit::add_op(
1717 const Op_ptr &op, const std::vector<ID> &args,
1718 std::optional<std::string> opgroup) {
1719 static_assert(std::is_base_of<UnitID, ID>::value);
1720 op_signature_t sig = op->get_signature();
1721
1722 // check if there are wasm or rng wires in the signature
1723 unsigned count_wasm_sig = 0;
1724 unsigned count_rng_sig = 0;
1725 for (EdgeType e : sig) {
1726 if (e == EdgeType::WASM) {
1727 ++count_wasm_sig;
1728 } else if (e == EdgeType::RNG) {
1729 ++count_rng_sig;
1730 }
1731 }
1732
1733 if (sig.size() != args.size()) {
1734 throw CircuitInvalidity(
1735 std::to_string(args.size()) + " args provided, but " + op->get_name() +
1736 " requires " + std::to_string(sig.size()));
1737 }
1738
1739 if (opgroup) {
1740 auto opgroupsig = opgroupsigs.find(opgroup.value());
1741 if (opgroupsig != opgroupsigs.end()) {
1742 if (opgroupsig->second != sig) {
1743 throw CircuitInvalidity("Mismatched signature for operation group");
1744 }
1745 } else {
1746 opgroupsigs[opgroup.value()] = sig;
1747 }
1748 }
1749
1750 add_wasm_register(count_wasm_sig);
1751 add_rng_register(count_rng_sig);
1752
1753 unit_set_t write_arg_set;
1754 EdgeVec preds;
1755 for (unsigned i = 0; i < args.size(); ++i) {
1756 const UnitID &arg = args[i];
1757 if (sig[i] != EdgeType::Boolean) {
1758 if (write_arg_set.find(arg) != write_arg_set.end())
1759 throw CircuitInvalidity(
1760 "Multiple operation arguments reference " + arg.repr());
1761 write_arg_set.insert(arg);
1762 }
1763 Vertex out_vert = get_out(arg);
1764 Edge pred_out_e = get_nth_in_edge(out_vert, 0);
1765 preds.push_back(pred_out_e);
1766 }
1767
1768 Vertex new_v;
1769 OpType optype = op->get_type();
1770 if (optype == OpType::CnRy && args.size() == 1) {
1771 new_v = add_vertex(
1772 get_op_ptr(OpType::Ry, as_gate_ptr(op)->get_params()), opgroup);
1773 } else if (optype == OpType::CnRx && args.size() == 1) {
1774 new_v = add_vertex(
1775 get_op_ptr(OpType::Rx, as_gate_ptr(op)->get_params()), opgroup);
1776 } else if (optype == OpType::CnRz && args.size() == 1) {
1777 new_v = add_vertex(
1778 get_op_ptr(OpType::Rz, as_gate_ptr(op)->get_params()), opgroup);
1779 } else if (optype == OpType::CnX && args.size() == 1) {
1780 new_v = add_vertex(get_op_ptr(OpType::X), opgroup);
1781 } else if (optype == OpType::CnZ && args.size() == 1) {
1782 new_v = add_vertex(get_op_ptr(OpType::Z), opgroup);
1783 } else if (optype == OpType::CnY && args.size() == 1) {
1784 new_v = add_vertex(get_op_ptr(OpType::Y), opgroup);
1785 } else {
1786 new_v = add_vertex(op, opgroup);
1787 }
1788 rewire(new_v, preds, sig);
1789 return new_v;
1790}
