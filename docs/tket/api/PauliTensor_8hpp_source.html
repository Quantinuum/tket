540template <typename PauliContainer, typename CoeffType>
541class PauliTensor {
542 static_assert(
543 std::is_same<PauliContainer, QubitPauliMap>::value ||
544 std::is_same<PauliContainer, DensePauliMap>::value,
545 "PauliTensor must be either dense or qubit-indexed.");
546 static_assert(
547 std::is_same<CoeffType, no_coeff_t>::value ||
548 std::is_same<CoeffType, quarter_turns_t>::value ||
549 std::is_same<CoeffType, Complex>::value ||
550 std::is_same<CoeffType, Expr>::value,
551 "PauliTensor must either support no coefficient, quarter turns, or "
552 "arbitrary complex (floating point or symbolic) coefficients.");
553
554 public:
555 PauliContainer string;
556 CoeffType coeff;
557
561 PauliTensor() : string(), coeff(default_coeff<CoeffType>()) {}
562
566 PauliTensor(
567 const PauliContainer &_string,
568 const CoeffType &_coeff = default_coeff<CoeffType>())
569 : string(_string), coeff(_coeff) {}

574 template <typename PC = PauliContainer>
575 PauliTensor(
576 const Qubit &q, Pauli p,
577 const CoeffType &_coeff = default_coeff<CoeffType>(),
578 typename std::enable_if<std::is_same<PC, QubitPauliMap>::value>::type * =
579 0)
580 : string({{q, p}}), coeff(_coeff) {}

586 template <typename PC = PauliContainer>
587 PauliTensor(
588 const DensePauliMap &_string,
589 const CoeffType &_coeff = default_coeff<CoeffType>(),
590 typename std::enable_if<std::is_same<PC, QubitPauliMap>::value>::type * =
591 0)
592 : string(cast_container<DensePauliMap, QubitPauliMap>(_string)),
593 coeff(_coeff) {}
594
599 template <typename PC = PauliContainer>
600 PauliTensor(
601 const std::list<Qubit> &qubits, const std::list<Pauli> &paulis,
602 const CoeffType &_coeff = default_coeff<CoeffType>(),
603 typename std::enable_if<std::is_same<PC, QubitPauliMap>::value>::type * =
604 0)
605 : string(), coeff(_coeff) {
606 if (qubits.size() != paulis.size()) {
607 throw std::logic_error(
608 "Mismatch of Qubits and Paulis upon QubitPauliString "
609 "construction");
610 }
611 std::list<Pauli>::const_iterator p_it = paulis.begin();
612 for (const Qubit &qb : qubits) {
613 Pauli p = *p_it;
614 if (string.find(qb) != string.end()) {
615 throw std::logic_error(
616 "Non-unique Qubit inserted into QubitPauliString map");
617 }
618 string[qb] = p;
619 ++p_it;
620 }
621 }

646 int compare(const PauliTensor<PauliContainer, CoeffType> &other) const {
647 int cont_comp =
648 compare_containers<PauliContainer>(this->string, other.string);
649 if (cont_comp != 0) return cont_comp;
650 return compare_coeffs<CoeffType>(this->coeff, other.coeff);
651 }
652 bool operator==(const PauliTensor<PauliContainer, CoeffType> &other) const {
653 return (compare(other) == 0);
654 }
655 bool operator!=(const PauliTensor<PauliContainer, CoeffType> &other) const {
656 return !(*this == other);
657 }
658 bool operator<(const PauliTensor<PauliContainer, CoeffType> &other) const {
659 return compare(other) < 0;
660 }

883 PauliTensor<PauliContainer, CoeffType> operator*(
884 const PauliTensor<PauliContainer, CoeffType> &other) const {
885 std::pair<quarter_turns_t, PauliContainer> prod =

891 }

984 Eigen::VectorXcd dot_state(const Eigen::VectorXcd &state) const {
985 // allowing room for big states
986 unsigned long long n = state.size();
987 if (!(n && (!(n & (n - 1)))))
988 throw std::logic_error("Statevector size is not a power of two.");
989 unsigned n_qubits = 0;
990 while (n) {
991 n >>= 1;
992 ++n_qubits;
993 }
994 --n_qubits;
995 return (to_sparse_matrix(n_qubits) * state);
996 }
997
1004 Eigen::VectorXcd dot_state(
1005 const Eigen::VectorXcd &state, const qubit_vector_t &qubits) const {
1006 if (state.size() != 1 << qubits.size())
1007 throw std::logic_error(
1008 "Size of statevector does not match number of qubits passed to "
1009 "dot_state");
1010 return (to_sparse_matrix(qubits) * state);
1011 }

1020 Complex state_expectation(const Eigen::VectorXcd &state) const {
1021 return state.dot(dot_state(state));
1022 }
